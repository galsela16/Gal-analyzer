cmake_minimum_required(VERSION 3.16)
project(GalNativeDSP LANGUAGES CXX)
add_library(GalDSP STATIC src/GalDSP.cpp src/GalGenerator.cpp)
target_include_directories(GalDSP PUBLIC include)
target_compile_features(GalDSP PUBLIC cxx_std_17)
add_executable(GalDSPTests tests/GalDSPTests.cpp)
target_link_libraries(GalDSPTests PRIVATE GalDSP)
enable_testing()
add_test(NAME NativeSignalAccuracy COMMAND GalDSPTests)

add_executable(GalGeneratorTests tests/GalGeneratorTests.cpp)
target_link_libraries(GalGeneratorTests PRIVATE GalDSP)
add_test(NAME NativeGeneratorAccuracy COMMAND GalGeneratorTests)
